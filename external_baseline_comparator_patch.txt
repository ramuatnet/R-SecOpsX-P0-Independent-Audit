baseline_summary_path = output_dir / "dependency_track_baseline" / "11_comparison_summary.json"
if baseline_summary_path.exists():
    _step("Generating external baseline comparison")
    try:
        run_external_baseline_comparison(
            dependency_track_summary_path=baseline_summary_path,
            rsecopsx_report_path=output_dir / "clean_report.json",
            output_dir=output_dir,
        )
        _safe_print(f"       => {output_dir / 'external_baseline_comparison.json'}")
        _safe_print(f"       => {output_dir / 'External_Baseline_Comparison.md'}")
    except Exception as exc:
        _safe_print(f"       => External baseline comparison skipped: {exc}")
